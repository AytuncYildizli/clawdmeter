# Firmware Implementation Plan — M5Stack Core 2

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Port the Clawdmeter firmware from the Waveshare ESP32-S3 + AMOLED + Linux daemon stack onto M5Stack Core 2 + macOS daemon, implementing the dual-provider Layout C design (Claude screen + Codex screen with swipe, auto-rotating big-number meter, splash overlay, agent-aware bezel buttons via Superset focus state).

**Architecture:** Two PlatformIO environments: `native` (host-side unit tests for JSON parsing, layout math, auto-rotate state, palette logic) and `m5stack-core2` (target firmware). M5Unified handles display, touch, PMU, IMU, and bezel buttons through a unified API. LVGL 9 renders the UI using a M5Unified framebuffer adapter. NimBLE-Arduino exposes the GATT service the daemon writes to, plus a separate BLE HID profile for bezel-button keystroke emission.

**Tech Stack:** C++17, PlatformIO, Arduino framework, M5Unified, LVGL 9, NimBLE-Arduino, ArduinoJson. Python 3.10+ for the splash-asset rescale pipeline.

### Task 3: Data structures

Defines the on-device representation of the BLE payload. No tests — it's pure data declaration; the parser test (Task 4) exercises the shape.

- [ ] **Step 1: Write `firmware/src/data.h`**

```cpp
#pragma once
#include <cstdint>

namespace data {

enum class AgentKind : uint8_t {
    None,
    Claude,
    Codex,
};

struct ProviderBlock {
    int s = 0;         // 5h utilization percent (0-100)
    int sr = 0;        // minutes until 5h reset
    int w = 0;         // 7d utilization percent
    int wr = 0;        // minutes until 7d reset
    char st[16] = {};  // "allow" | "warn" | "block" | "unknown" | "unavailable"
    bool ok = false;
};

struct FocusBlock {
    AgentKind agent = AgentKind::None;
    char repo[32] = {};   // worktree-derived project name
    int sessions = 0;     // count of other active agent panes
};

struct PayloadState {
    ProviderBlock claude;
    ProviderBlock codex;
    FocusBlock focus;
    bool initialized = false;  // becomes true after the first successful parse
};

}  // namespace data
```

- [ ] **Step 4: Write `firmware/src/payload_parser.cpp`**

```cpp
#include "payload_parser.h"
#include <ArduinoJson.h>
#include <cstring>

namespace parser {

static void copy_safe(char* dst, size_t cap, const char* src) {
    if (!src) {
        dst[0] = '\0';
        return;
    }
    std::strncpy(dst, src, cap - 1);
    dst[cap - 1] = '\0';
}

static data::AgentKind classify(const char* s) {
    if (!s) return data::AgentKind::None;
    if (std::strcmp(s, "claude") == 0) return data::AgentKind::Claude;
    if (std::strcmp(s, "codex") == 0) return data::AgentKind::Codex;
    return data::AgentKind::None;
}

static void parse_provider(JsonObjectConst obj, data::ProviderBlock& out) {
    out.s  = obj["s"]  | 0;
    out.sr = obj["sr"] | 0;
    out.w  = obj["w"]  | 0;
    out.wr = obj["wr"] | 0;
    out.ok = obj["ok"] | false;
    copy_safe(out.st, sizeof(out.st), obj["st"] | "unknown");
}

bool parse_payload(const char* json, size_t len, data::PayloadState& out) {
    JsonDocument doc;
    DeserializationError err = deserializeJson(doc, json, len);
    if (err) {
        return false;
    }
    parse_provider(doc["claude"].as<JsonObjectConst>(), out.claude);
    parse_provider(doc["codex"].as<JsonObjectConst>(), out.codex);

    auto focus = doc["focus"].as<JsonObjectConst>();
    out.focus.agent = classify(focus["agent"] | "none");
    copy_safe(out.focus.repo, sizeof(out.focus.repo), focus["repo"] | "");
    out.focus.sessions = focus["sessions"] | 0;

    out.initialized = true;
    return true;
}

}  // namespace parser
```

#pragma once
#include <cstdint>

namespace rotate {

enum class Frame : uint8_t { FiveHour, SevenDay };

struct State {
    uint32_t epoch_ms = 0;  // ms-of-day anchor for the cycle
};

void reset(State& st, uint32_t now_ms);
Frame current(const State& st, uint32_t now_ms);

}  // namespace rotate

- [ ] **Step 4: Write `firmware/src/auto_rotate.cpp`**

```cpp
#include "auto_rotate.h"

namespace rotate {

constexpr uint32_t PERIOD_MS = 5000;

void reset(State& st, uint32_t now_ms) {
    st.epoch_ms = now_ms;
}

Frame current(const State& st, uint32_t now_ms) {
    uint32_t elapsed = now_ms - st.epoch_ms;  // overflow-safe via unsigned arithmetic
    uint32_t cycle_pos = (elapsed / PERIOD_MS) % 2;
    return (cycle_pos == 0) ? Frame::FiveHour : Frame::SevenDay;
}

}  // namespace rotate
```

constexpr const char* SERVICE_UUID = "4c41555a-4465-7669-6365-000000000001";

- [ ] **Step 2: Write `firmware/src/ble_peer.cpp`**

```cpp
#include "ble_peer.h"
#include "payload_parser.h"
#include <NimBLEDevice.h>

namespace ble_peer {

static NimBLEServer* g_server = nullptr;
static NimBLECharacteristic* g_req_char = nullptr;
static PayloadCallback g_callback;

class RxCallbacks : public NimBLECharacteristicCallbacks {
    void onWrite(NimBLECharacteristic* chr, NimBLEConnInfo& /*info*/) override {
        const auto& value = chr->getValue();
        data::PayloadState parsed{};
        if (parser::parse_payload(value.c_str(), value.length(), parsed)) {
            if (g_callback) g_callback(parsed);
        }
    }
};

void begin(PayloadCallback cb) {
    g_callback = cb;

    NimBLEDevice::init(DEVICE_NAME);
    NimBLEDevice::setMTU(517);  // request large MTU for payload writes

    g_server = NimBLEDevice::createServer();
    NimBLEService* service = g_server->createService(SERVICE_UUID);

    auto rx = service->createCharacteristic(
        RX_CHAR_UUID,
        NIMBLE_PROPERTY::WRITE | NIMBLE_PROPERTY::WRITE_NR
    );
    static RxCallbacks rx_cb;
    rx->setCallbacks(&rx_cb);

    g_req_char = service->createCharacteristic(
        REQ_CHAR_UUID,
        NIMBLE_PROPERTY::NOTIFY | NIMBLE_PROPERTY::READ
    );

    service->start();

    NimBLEAdvertising* adv = NimBLEDevice::getAdvertising();
    adv->addServiceUUID(SERVICE_UUID);
    adv->setName(DEVICE_NAME);
    adv->start();
}

void request_refresh() {
    if (g_req_char) {
        const uint8_t payload[] = { 'r' };
        g_req_char->setValue(payload, sizeof(payload));
        g_req_char->notify();
    }
}

void tick() {
    // NimBLE handles its own event pump; nothing required here for now.
}

}  // namespace ble_peer

    // Configure manufacturer, PnP IDs, etc.
    g_hid->setManufacturer("Clawdmeter");
    g_hid->setPnp(0x02, 0xe502, 0xa111, 0x0210);
    g_hid->setHidInfo(0x00, 0x01);
    // Set HID report descriptor (boot keyboard) — full bytes from NimBLE example
    static const uint8_t REPORT_MAP[] = { /* boot keyboard descriptor */ };
    g_hid->setReportMap((uint8_t*)REPORT_MAP, sizeof(REPORT_MAP));
    g_hid->startServices();
}

void send_key(Modifier mod, Key key) {
    if (!g_input) return;
    uint8_t report[8] = { (uint8_t)mod, 0, (uint8_t)key, 0, 0, 0, 0, 0 };
    g_input->setValue(report, 8);
    g_input->notify();
    // Send key-up
    uint8_t release[8] = { 0 };
    g_input->setValue(release, 8);
    g_input->notify();
}

}  // namespace ble_hid
```

The `REPORT_MAP` boot-keyboard descriptor is standard — fetch from NimBLE-Arduino's `HID_keyboard.ino` example. Don't hand-craft from spec; it's ~70 bytes of HID descriptor magic.

```python
"""Rescale upstream 480×480 Clawd sprites to 320×240 with nearest-neighbor.
Outputs C array headers consumable by LVGL's lv_img_dsc_t.

Usage:
    python rescale_sprites.py --in research/upstream-assets --out firmware/src/assets
"""
from __future__ import annotations
import argparse
from pathlib import Path
import struct

try:
    from PIL import Image
except ImportError:
    raise SystemExit("install Pillow: pip install Pillow")


def to_rgb565(rgb: tuple[int, int, int]) -> int:
    r, g, b = rgb
    return ((r & 0xF8) << 8) | ((g & 0xFC) << 3) | (b >> 3)


def emit_header(img: Image.Image, name: str, out_path: Path) -> None:
    img = img.convert("RGB")
    pixels = []
    for y in range(img.height):
        for x in range(img.width):
            pixels.append(to_rgb565(img.getpixel((x, y))))
    raw = b"".join(struct.pack("<H", p) for p in pixels)
    hex_lines = []
    for i in range(0, len(raw), 16):
        chunk = raw[i:i + 16]
        hex_lines.append(", ".join(f"0x{b:02x}" for b in chunk))
    body = ",\n    ".join(hex_lines)
    out_path.write_text(f"""// Auto-generated by tools/rescale_sprites.py — do not edit
#pragma once
#include <cstdint>

namespace assets {{
constexpr int {name.upper()}_W = {img.width};
constexpr int {name.upper()}_H = {img.height};
constexpr uint8_t {name.upper()}_DATA[] = {{
    {body}
}};
}}
""")


def main() -> int:
    ap = argparse.ArgumentParser()
    ap.add_argument("--in", dest="in_dir", required=True)
    ap.add_argument("--out", dest="out_dir", required=True)
    ap.add_argument("--target", default="320x240")
    args = ap.parse_args()

    in_dir = Path(args.in_dir)
    out_dir = Path(args.out_dir)
    out_dir.mkdir(parents=True, exist_ok=True)

    tw, th = map(int, args.target.split("x"))
    for src in in_dir.glob("clawd*.png"):
        img = Image.open(src).resize((tw, th), Image.NEAREST)
        emit_header(img, src.stem, out_dir / f"{src.stem}.h")
        print(f"wrote {out_dir / (src.stem + '.h')}")
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
```
